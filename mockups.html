ЧИСТАЯ РЕАЛИЗАЦИЯ (impl/guards.js):
canSwitchTab: (state, bag, world) => {
  const target = bag.action && bag.action.tab;
  if (target === 'model') return true;
  if (state.open.kind === 'assembly') return false;
  if (!state.open.path) return false;
  return (world.drawings || []).includes(state.open.path);
}
